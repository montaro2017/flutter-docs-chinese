使此 CancelableCompleter.operation 以与 result 相同的结果完成。

如果 propagateCancel 为 true（默认值），并且此 completer 的 operation 在 result 完成之前被取消，那么 result 也会被取消。

实现

void completeOperation(
  CancelableOperation<T> result, {
  bool propagateCancel = true,
}) {
  if (!_mayComplete) throw StateError('Already completed');
  _mayComplete = false;
  if (isCanceled) {
    if (propagateCancel) result.cancel();
    result.value.ignore();
    return;
  }
  result.then<void>(
    (value) {
      _inner?.complete(
        value,
      ); // _inner is set to null if this.operation is cancelled.
    },
    onError: (error, stack) {
      _inner?.completeError(error, stack);
    },
    onCancel: () {
      operation.cancel();
    },
  );
  if (propagateCancel) {
    _cancelCompleter?.future.whenComplete(result.cancel);
  }
}
