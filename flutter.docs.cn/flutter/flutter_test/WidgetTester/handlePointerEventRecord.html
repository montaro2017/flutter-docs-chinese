模拟框架处理从 Flutter 引擎接收的一系列 PointerEvent。

PointerEventRecord.timeDelay 用作事件注入相对于此方法调用起点的时间延迟。

返回一个列表，其中包含处理 PointerEventRecord.events 时的实际延迟时间与 PointerEventRecord.timeDelay 之间的差值。

  • 对于 AutomatedTestWidgetsFlutterBinding，当时钟为虚拟时，该返回值应精确为零。
  • 对于 LiveTestWidgetsFlutterBinding，这些数值通常是较小的正数，表示事件发生时间略晚于设定时间；少数情况下也可能出现几十微秒左右的微小负值。这是 Future.delayed 的特性所致。

返回值越接近零，执行结果就越忠实于 records。

另请参阅 PointerEventRecord。

实现

@override
Future<List<Duration>> handlePointerEventRecord(Iterable<PointerEventRecord> records) {
  assert(records.isNotEmpty);
  return TestAsyncUtils.guard<List<Duration>>(() async {
    final handleTimeStampDiff = <Duration>[];
    DateTime? startTime;
    for (final record in records) {
      final DateTime now = binding.clock.now();
      startTime ??= now;
      // So that the first event is promised to receive a zero timeDiff
      final Duration timeDiff = record.timeDelay - now.difference(startTime);
      if (timeDiff.isNegative) {
        // Flush all past events
        handleTimeStampDiff.add(-timeDiff);
        for (final PointerEvent event in record.events) {
          binding.handlePointerEventForSource(event, source: TestBindingEventSource.test);
        }
      } else {
        await binding.pump();
        await binding.delayed(timeDiff);
        handleTimeStampDiff.add(binding.clock.now().difference(startTime) - record.timeDelay);
        for (final PointerEvent event in record.events) {
          binding.handlePointerEventForSource(event, source: TestBindingEventSource.test);
        }
      }
    }
    await binding.pump();
    // This makes sure that a gesture is completed, with no more pointers
    // active.
    return handleTimeStampDiff;
  });
}
