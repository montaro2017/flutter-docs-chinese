static void startSync(String name, {Map? arguments, Flow? flow}) {
  if (!_hasTimeline) return;
  if (!_isDartStreamEnabled()) {
    // Push a null onto the stack and return.
    _stack.add(null);
    return;
  }
  var block = new _SyncBlock._(
    name,
    _getNextTaskId(),
    arguments: arguments,
    flow: flow,
  );
  _stack.add(block);
  block._startSync();
}
