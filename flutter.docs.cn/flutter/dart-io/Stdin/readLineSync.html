从 stdin 读取一行。

此调用会一直阻塞，直到有完整的一行可读取。

行结束符可以是 <CR><LF> 或 <LF>。在 Windows 上，如果 stdin 的 stdioType 为 StdioType.terminal，结束符也可以是单独的 <CR>。

输入字节会使用 encoding 转换为字符串。如果省略 encoding，则默认使用 systemEncoding。

如果 retainNewlines 为 false，返回的字符串不包含末尾的行结束符；如果为 true，则包含行结束符。默认值为 false。

如果读取 stdin 的若干字节后到达文件末尾，会返回这些数据，且不附带行结束符。如果输入结束前没有读取到任何字节，则返回 null。

实现

String? readLineSync({
  Encoding encoding = systemEncoding,
  bool retainNewlines = false,
}) {
  const CR = 13;
  const LF = 10;
  final List<int> line = <int>[];
  // On Windows, if lineMode is disabled, only CR is received.
  bool crIsNewline =
      Platform.isWindows &&
      (stdioType(stdin) == StdioType.terminal) &&
      !lineMode;
  if (retainNewlines) {
    int byte;
    do {
      byte = readByteSync();
      if (byte < 0) {
        break;
      }
      line.add(byte);
    } while (byte != LF && !(byte == CR && crIsNewline));
    if (line.isEmpty) {
      return null;
    }
  } else if (crIsNewline) {
    // CR and LF are both line terminators, neither is retained.
    while (true) {
      int byte = readByteSync();
      if (byte < 0) {
        if (line.isEmpty) return null;
        break;
      }
      if (byte == LF || byte == CR) break;
      line.add(byte);
    }
  } else {
    // Case having to handle CR LF as a single unretained line terminator.
    outer:
    while (true) {
      int byte = readByteSync();
      if (byte == LF) break;
      if (byte == CR) {
        do {
          byte = readByteSync();
          if (byte == LF) break outer;

          line.add(CR);
        } while (byte == CR);
        // Fall through and handle non-CR character.
      }
      if (byte < 0) {
        if (line.isEmpty) return null;
        break;
      }
      line.add(byte);
    }
  }
  return encoding.decode(line);
}
