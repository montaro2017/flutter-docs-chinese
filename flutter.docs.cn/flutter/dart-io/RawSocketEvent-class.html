RawSocketEvent 类

RawDatagramSocket、RawSecureSocket 和 RawSocket 的事件。

当套接字状态发生变化时，套接字会通过 Stream 行为使用这些事件对象（例如 RawSocket.listen、RawSocket.forEach）。

import 'dart:convert';
import 'dart:io';

void main() async {
  final socket = await RawSocket.connect("example.com", 80);

  socket.listen((event) {
    switch (event) {
      case RawSocketEvent.read:
        final data = socket.read();
        if (data != null) {
          print(ascii.decode(data));
        }
        break;
      case RawSocketEvent.write:
        socket.write(ascii.encode('GET /\r\nHost: example.com\r\n\r\n'));
        socket.writeEventsEnabled = false;
        break;
      case RawSocketEvent.readClosed:
        socket.close();
        break;
      case RawSocketEvent.closed:
        break;
      default:
        throw "Unexpected event $event";
    }
  });
}
