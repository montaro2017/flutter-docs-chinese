以编程方式创建 KeyboardEvent。

由于浏览器之间存在差异，此基础级构造函数无法指定 keyCode、charCode 或 keyIdentifier 值。它允许用户以编程方式创建并分发 KeyboardEvent，但事件不会包含任何特定按键的信息。如需创建包含特定按键值的键盘事件，请参阅自定义事件 KeyEvent。

实现

factory KeyboardEvent(
  String type, {
  Window? view,
  bool canBubble = true,
  bool cancelable = true,
  int? location,
  int? keyLocation, // Legacy alias for location
  bool ctrlKey = false,
  bool altKey = false,
  bool shiftKey = false,
  bool metaKey = false,
}) {
  if (view == null) {
    view = window;
  }
  location ??= keyLocation ?? 1;
  KeyboardEvent e = document._createEvent("KeyboardEvent") as KeyboardEvent;
  e._initKeyboardEvent(
    type,
    canBubble,
    cancelable,
    view,
    "",
    location,
    ctrlKey,
    altKey,
    shiftKey,
    metaKey,
  );
  return e;
}
