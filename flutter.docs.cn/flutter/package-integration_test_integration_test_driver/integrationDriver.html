使用 flutter drive 运行集成测试的适配器。 若要使用 flutter drive 运行集成测试 <test_name>.dart，请在应用的 test_driver 目录中放置名为 <test_name>_test.dart 的文件： 
import 'dart:async';

import 'package:integration_test/integration_test_driver.dart';

Future<void> main() async => integrationDriver();

参数：

timeout 用于限制测试结束前的最长等待时间。它不一定等于测试应用的运行时长；测试可能会在 timeout 期限之前结束。 responseDataCallback 是处理 Response.data 的回调。默认值为 writeResponseData。 writeResponseOnFailure 用于决定测试失败时是否调用 responseDataCallback 函数来处理 Response.data。默认值为 false。

实现

Future<void> integrationDriver({
  Duration timeout = const Duration(minutes: 20),
  ResponseDataCallback? responseDataCallback = writeResponseData,
  bool writeResponseOnFailure = false,
}) async {
  final FlutterDriver driver = await FlutterDriver.connect();
  final String jsonResult = await driver.requestData(null, timeout: timeout);
  final Response response = Response.fromJson(jsonResult);

  await driver.close();

  if (response.allTestsPassed) {
    print('All tests passed.');
    if (responseDataCallback != null) {
      await responseDataCallback(response.data);
    }
    exit(0);
  } else {
    print('Failure Details:\n${response.formattedFailureDetails}');
    if (responseDataCallback != null && writeResponseOnFailure) {
      await responseDataCallback(response.data);
    }
    exit(1);
  }
}
